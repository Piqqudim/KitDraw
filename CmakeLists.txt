cmake_minimum_required(VERSION 3.50)

project(KitDraw VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  SET(CMAKE_BUILD_TYPE "Release" CACHE STRING "Build Type" FORCE)
endif()

add_executable(KitDraw 
               src/main.cpp 
                     )
target_include_directories(KitDraw PUBLIC 
                           include/Gui
                           include/Renderer
                           ThirdParty
                           ThirdParty/glbinding/include
                           ThirdParty/glbinding/include/glbinding
                           ThirdParty/imgui
                           ThirdParty/imgui/backends
)

target_link_libraries(KitDraw PUBLIC glbinding.dll glfw3dll imgui.dll opengl32)
target_link_directories(KitDraw PUBLIC lib)
